tmux new -s vllm                      # persistent session
export MODEL_API_KEY="abc-123"        # your endpoint token
export HF_HOME=/path/to/shared/cache  # <-- confirm the platform's shared-cache path


VLLM_USE_TRITON_FLASH_ATTN=0 \

  --trust-remote-code \
  --max_model_len 24272 \
